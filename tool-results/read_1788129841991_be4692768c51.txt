   156→// Seasonal / weekend rate adjustments — applied on top of base price
   157→model SeasonalRate {
   158→  id              String   @id @default(cuid())
   159→  nameAr          String
   160→  nameEn          String
   161→  roomTypeId      String?  // null = applies to all room types
   162→  roomType        RoomType? @relation(fields: [roomTypeId], references: [id], onDelete: Cascade)
   163→  ratePlanId      String?  // null = applies to all rate plans
   164→  ratePlan        RatePlan? @relation(fields: [ratePlanId], references: [id], onDelete: Cascade)
   165→  startDate       DateTime // season start (inclusive)
   166→  endDate         DateTime // season end (inclusive)
   167→  adjustmentType  String   @default("PERCENTAGE") // PERCENTAGE or FIXED
   168→  adjustmentValue Float    // e.g. 20 = +20%, -10 = -10%, or fixed amount
   169→  // Weekend surcharge (applied on Fri/Sat for Yemen weekend)
   170→  weekendAdjustmentType String? // PERCENTAGE or FIXED
   171→  weekendAdjustmentValue Float?
   172→  weekendDays     String   @default("5,6") // 0=Sun,5=Fri,6=Sat (comma-separated)
   173→  isActive        Boolean  @default(true)
   174→  priority        Int      @default(0) // higher priority wins when overlapping
   175→  minStayNights   Int?     // optional: override hotel min stay during this season
   176→  maxStayNights   Int?     // optional: override hotel max stay during this season
   177→  createdAt       DateTime @default(now())
   178→  updatedAt       DateTime @updatedAt
   179→
   180→  @@index([roomTypeId])
   181→  @@index([ratePlanId])
   182→  @@index([startDate, endDate])
   183→  @@index([isActive])
   184→}
